const gretzingerPlinkData = await (async () => {
  const bimDat = await FileAttachment("data/2024_Gretzinger_EarlyCelts.bim").text().then(t => readBimData(t));
  const famDat = await FileAttachment("data/2024_Gretzinger_EarlyCelts.fam").text().then(t => readFamData(t));

})()

### Reducing the genotype data to the overlap

Testing the PCA projection

const params = display(await FileAttachment("data/Joscha_HiRes_WestEurasia_parameters.json").json())
const projectionResults = display(projectSamples(genoMatrix, reducedPcWeights, reducedFrequencies, plinkDat.numInds, weightDat.numPCs, params.nSNPs, params.yScale, params.eigenValues))
const projResultTable = projectionResults.map(o => ({nonMissing: o.nonMissingCount, pc1: o.pcCoordinates[0], pc2: o.pcCoordinates[1]}))
display(Inputs.table(projResultTable))
